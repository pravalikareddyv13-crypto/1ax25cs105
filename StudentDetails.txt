Program 3:
Create a Java class called Student with the USN, Name, Branch, Phone details as variables within it. Write a 
Java program to create N Student objects and print the USN, Name, Branch, and Phone of these objects with 
suitable headings.

import java.util.Scanner;
class Student
{
    private String usn;
    private String name;
    private String branch;
    private String phone;

        // Constructor to initialize student details
        public Student(String usn, String name, String branch, String phone) 
        {
            this.usn = usn;
            this.name = name;
            this.branch = branch;
            this.phone = phone;
        }

    // Method to display student details in tabular format
        public void display() 
        {
        System.out.printf("%-15s %-20s %-15s %-15s\n", usn, name, branch, phone);
        }
}

// Main class
public class StudentDetails 
{
    public static void main(String[] args) 
    {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter the number of students (N): ");
        int n = sc.nextInt();
        sc.nextLine(); 

        // Array to store N Student objects
        Student[] s = new Student[n];

        // Input details for N students
        for (int i = 0; i < n; i++) 
        {
            System.out.println("\nEnter details for Student " + (i + 1) + ":");
            
            System.out.print("USN: ");
            String usn = sc.nextLine();

            System.out.print("Name: ");
            String name = sc.nextLine();

            System.out.print("Branch: ");
            String branch = sc.nextLine();

            System.out.print("Phone: ");
            String phone = sc.nextLine();

            s[i] = new Student(usn, name, branch, phone);
        }

        // Display output with headings
        System.out.println("\n---------------------------------------------------------------");
        System.out.printf("%-15s %-20s %-15s %-15s\n", "USN", "Name", "Branch", "Phone");
        System.out.println("-----------------------------------------------------------------");

        for (int i = 0; i < n; i++)
        {
            s[i].display();
        }
        System.out.println("-----------------------------------------------------------------");
        sc.close();
    }
}
